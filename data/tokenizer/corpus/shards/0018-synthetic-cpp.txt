#include <algorithm>
#include <vector>
#include <string>
#include <unordered_map>

template <typename T>
class Pool {
public:
    T& acquire() {
        if (free_.empty()) return storage_.emplace_back();
        T& slot = *free_.back();
        free_.pop_back();
        return slot;
    }
    void release(T& obj) { free_.push_back(&obj); }
private:
    std::vector<T> storage_;
    std::vector<T*> free_;
};

int count_words(const std::string& text) {
    std::unordered_map<std::string, int> freq;
    std::string token;
    for (char c : text) {
        if (std::isspace(static_cast<unsigned char>(c))) {
            if (!token.empty()) { ++freq[token]; token.clear(); }
        } else {
            token.push_back(c);
        }
    }
    if (!token.empty()) ++freq[token];
    return static_cast<int>(freq.size());
}

#include <unordered_map>

template <typename T>
class Pool {
public:
    T& acquire() {
        if (free_.empty()) return storage_.emplace_back();
        T& slot = *free_.back();
        free_.pop_back();
        return slot;
    }
    void release(T& obj) { free_.push_back(&obj); }
private:
    std::vector<T> storage_;
    std::vector<T*> free_;
};

int count_words(const std::string& text) {
    std::unordered_map<std::string, int> freq;
    std::string token;
    for (char c : text) {
        if (std::isspace(static_cast<unsigned char>(c))) {
            if (!token.empty()) { ++freq[token]; token.clear(); }
        } else {
            token.push_back(c);
        }
    }
    if (!token.empty()) ++freq[token];
    return static_cast<int>(freq.size());
}
